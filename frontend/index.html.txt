            <div class="text-section-title">Up Next</div>
            <div id="up-next">Loading...</div>
        </div>
    </div>

    <script>
        const videoPlayer = document.getElementById("video-player");
        const showingNowText = document.getElementById("showing-now");
        const upNextText = document.getElementById("up-next");

        const apiUrl = "https://reikertv.onrender.com/api/get-current";
        const saveProgressUrl = "https://reikertv.onrender.com/api/save-progress";

        let isSwitching = false;

        // Fetch movie data and update UI
        async function fetchMovieData() {
            try {
                const response = await fetch(apiUrl);
                const data = await response.json();

                // Update UI with current and up-next movie details
                showingNowText.innerText = data.showing_now.name || "No Movie Playing";
                upNextText.innerText = data.up_next.name || "No Movie Scheduled";

                if (!isSwitching) {
                    playVideo(data.showing_now, data.current_time);
                }
            } catch (error) {
                console.error("Error fetching movie data:", error);
            }
        }

        // Play video from the calculated current time
        function playVideo(videoData, currentTime) {
            if (videoData.video_url) {
                videoPlayer.src = videoData.video_url;
                videoPlayer.currentTime = Math.max(0, currentTime);
                videoPlayer.play();
            } else {
                console.log("No video URL provided.");
            }
        }

        // Save playback progress periodically
        async function savePlaybackProgress() {
            try {
                const progress = {
                    video_url: videoPlayer.src,
                    current_time: videoPlayer.currentTime
                };
                await fetch(saveProgressUrl, {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json"
                    },
                    body: JSON.stringify(progress)
                });
            } catch (error) {
                console.error("Error saving playback progress:", error);
            }
        }

        // Save playback progress when time updates
        videoPlayer.addEventListener("timeupdate", () => {
            savePlaybackProgress();
        });

        // Play the "Up Next" video when the current one ends
        videoPlayer.addEventListener("ended", async () => {
            isSwitching = true;
            try {
                const response = await fetch(apiUrl);
                const data = await response.json();

                showingNowText.innerText = data.up_next.name || "No Movie Scheduled";
                upNextText.innerText = "No Movie Scheduled";

                playVideo(data.up_next, 0);
            } catch (error) {
                console.error("Error playing up-next video:", error);
            }
            isSwitching = false;
        });

        // Initialize video playback on page load
        fetchMovieData();

        // Refresh movie data periodically (e.g., every 30 seconds)
        setInterval(fetchMovieData, 300000);
    </script>
